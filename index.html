-- ===================================================================== -- Olimpíadas IN-NOVA · Cronograma à prova de conflitos -- Rode no SQL Editor do Supabase. Pode rodar de novo sem problema. -- -- * O banco recusa qualquer horário que coloque, ao mesmo tempo: -- - duas atividades da MESMA CATEGORIA (exceto quando confirmado no -- cronograma manual: coluna permite_cat); -- - duas atividades no MESMO LOCAL; -- - um mesmo ÁRBITRO em duas atividades. -- Vale para o cronograma manual e para o automático. -- * Uma trava garante isso mesmo com várias pessoas salvando ao mesmo -- tempo: as gravações do cronograma entram uma de cada vez. -- * salvar_agenda(): o gerador automático grava tudo numa transação só. -- Se qualquer atividade chocar, nada é gravado. -- ===================================================================== -- autorização para a mesma categoria ao mesmo tempo (só pelo modo manual, com confirmação) alter table public.agenda add column if not exists permite_cat boolean not null default false; -- minutos desde 00:00 a partir de "HH:MM" (aceita coluna texto ou time) create or replace function public.ag_min(h text) returns int language sql immutable as $$ select case when coalesce(h, '') ~ '^\d{1,2}:\d{2}' then split_part(h, ':', 1)::int * 60 + split_part(h, ':', 2)::int end; $$; create or replace function public.ag_arbs(a text) returns text[] language sql immutable as $$ select coalesce(array_remove(string_to_array(replace(coalesce(a, ''), ' ', ''), ','), ''), '{}'); $$; create or replace function public.agenda_valida() returns trigger language plpgsql security definer set search_path = public as $$ declare ini int := public.ag_min(new.hr::text); fim int; c record; begin if new.dt is null or ini is null then return new; end if; -- sem horário: nada a conferir perform pg_advisory_xact_lock(hashtext('olimpiadas-agenda')); -- uma gravação do cronograma por vez fim := ini + coalesce(nullif(new.dur::int, 0), 60); select a.id, a.hr::text as hr, a.cat::text as cat, a.loc, (a.cat::text = new.cat::text and not coalesce(new.permite_cat, false)) as mesma_cat, (coalesce(new.loc, '') <> '' and a.loc = new.loc) as mesmo_loc into c from public.agenda a where a.id <> new.id and a.dt = new.dt and public.ag_min(a.hr::text) is not null and public.ag_min(a.hr::text) < fim and public.ag_min(a.hr::text) + coalesce(nullif(a.dur::int, 0), 60) > ini and ((a.cat::text = new.cat::text and not coalesce(new.permite_cat, false)) or (coalesce(new.loc, '') <> '' and a.loc = new.loc) or public.ag_arbs(a.arb) && public.ag_arbs(new.arb)) limit 1; if found then raise exception 'CONFLITO_AGENDA: % já está em % às % (% em uso)', split_part(c.id, '_', 1), new.dt, left(c.hr, 5), case when c.mesma_cat then 'categoria' when c.mesmo_loc then 'local' else 'árbitro' end using errcode = 'P0001'; end if; return new; end $$; drop trigger if exists agenda_valida on public.agenda; create trigger agenda_valida before insert or update on public.agenda for each row execute function public.agenda_valida(); -- índice para a conferência ser rápida mesmo com o cronograma cheio create index if not exists agenda_dt on public.agenda (dt); -- Gravação em lote do gerador automático (tudo ou nada) create or replace function public.salvar_agenda(linhas jsonb) returns int language plpgsql security invoker set search_path = public as $$ declare n int; begin if not public.pode_editar() then raise exception 'sem permissão' using errcode = '42501'; end if; perform pg_advisory_xact_lock(hashtext('olimpiadas-agenda')); -- libera os horários antigos das atividades que vão mudar (evita conflito com a posição antiga) update public.agenda set dt = null, hr = null where id in (select x->>'id' from jsonb_array_elements(linhas) x); insert into public.agenda (id, mod, cat, dt, hr, dur, loc, arb, permite_cat, updated_at) select r.id, r.mod, r.cat, r.dt, r.hr, r.dur, r.loc, r.arb, false, now() from jsonb_populate_recordset(null::public.agenda, linhas) r on conflict (id) do update set mod = excluded.mod, cat = excluded.cat, dt = excluded.dt, hr = excluded.hr, dur = excluded.dur, loc = excluded.loc, arb = excluded.arb, permite_cat = false, updated_at = now(); get diagnostics n = row_count; return n; end $$; revoke all on function public.salvar_agenda(jsonb) from public, anon; grant execute on function public.salvar_agenda(jsonb) to authenticated; -- Conflitos que JÁ existem no cronograma (para conferir antes): -- select a.id, b.id, a.dt, a.hr, b.hr from public.agenda a join public.agenda b -- on a.id < b.id and a.dt = b.dt -- and public.ag_min(a.hr::text) < public.ag_min(b.hr::text) + coalesce(nullif(b.dur::int,0),60) -- and public.ag_min(b.hr::text) < public.ag_min(a.hr::text) + coalesce(nullif(a.dur::int,0),60) -- and (a.cat::text = b.cat::text or (coalesce(a.loc,'') <> '' and a.loc = b.loc) -- or public.ag_arbs(a.arb) && public.ag_arbs(b.arb));
