Pode rodar de novo -- sem problema (o script é idempotente). -- -- Regras: -- * Qualquer pessoa (sem login) só LÊ os dados: placar, modalidades, -- integrantes, cronograma, regulamentos. -- * Só contas "editor" ou "admin" GRAVAM (inserir, alterar, apagar). -- * Conta nova nasce "pendente" (sem permissão) até o admin liberar. -- * mrcampeonato@gmail.com é sempre administrador (quando entra pelo -- Google, que garante que a pessoa é dona do email) e não pode ser -- rebaixado pela tela de contas. -- ===================================================================== -- 1. Perfis (um por conta de login) ----------------------------------- create table if not exists public.perfis ( id uuid primary key references auth.users(id) on delete cascade, email text not null, nome text not null default '', papel text not null default 'pendente' check (papel in ('pendente','editor','admin')), criado_em timestamptz not null default now() ); alter table public.perfis enable row level security; -- 2. Funções de permissão (rápidas: consulta pela chave primária) ----- create or replace function public.pode_editar() returns boolean language sql stable security definer set search_path = public as $$ select exists (select 1 from public.perfis where id = auth.uid() and papel in ('editor','admin')); $$; create or replace function public.eh_admin() returns boolean language sql stable security definer set search_path = public as $$ select exists (select 1 from public.perfis where id = auth.uid() and papel = 'admin'); $$; -- 3. Perfil criado automaticamente a cada cadastro -------------------- create or replace function public.novo_perfil() returns trigger language plpgsql security definer set search_path = public as $$ begin insert into public.perfis (id, email, nome, papel) values ( new.id, lower(new.email), left(coalesce(new.raw_user_meta_data->>'full_name', new.raw_user_meta_data->>'name', new.raw_user_meta_data->>'nome', ''), 60), case when lower(new.email) = 'mrcampeonato@gmail.com' and new.raw_app_meta_data->>'provider' = 'google' then 'admin' else 'pendente' end ) on conflict (id) do nothing; return new; end $$; drop trigger if exists ao_criar_usuario on auth.users; create trigger ao_criar_usuario after insert on auth.users for each row execute function public.novo_perfil(); -- Contas que já existiam antes deste script insert into public.perfis (id, email, nome, papel) select u.id, lower(u.email), left(coalesce(u.raw_user_meta_data->>'full_name', u.raw_user_meta_data->>'name', u.raw_user_meta_data->>'nome', ''), 60), case when lower(u.email) = 'mrcampeonato@gmail.com' and u.raw_app_meta_data->>'provider' = 'google' then 'admin' else 'pendente' end from auth.users u on conflict (id) do update set papel = case when excluded.papel = 'admin' then 'admin' else public.perfis.papel end; -- 4. Quem vê os perfis: cada um o seu; o admin vê todos --------------- drop policy if exists "perfil proprio ou admin" on public.perfis; create policy "perfil proprio ou admin" on public.perfis for select to authenticated using (id = auth.uid() or (select public.eh_admin())); -- Sem políticas de insert/update/delete: ninguém altera perfis direto. -- A troca de acesso passa só pela função abaixo. -- 5. Troca de acesso (só admin) ---------------------------------------- create or replace function public.definir_papel(alvo uuid, novo text) returns void language plpgsql security definer set search_path = public as $$ begin if not public.eh_admin() then raise exception 'sem permissão' using errcode = '42501'; end if; if novo not in ('pendente','editor','admin') then raise exception 'acesso inválido'; end if; if exists (select 1 from public.perfis where id = alvo and email = 'mrcampeonato@gmail.com') then raise exception 'O administrador principal não pode ser alterado'; end if; update public.perfis set papel = novo where id = alvo; end $$; revoke all on function public.definir_papel(uuid, text) from public, anon; grant execute on function public.definir_papel(uuid, text) to authenticated; -- 6. Tabelas do site: leitura livre, gravação só para editor/admin ----- do $$ declare t text; r record; begin foreach t in array array['cfg','res','reg','membros','inscricoes'] loop if to_regclass('public.' || t) is null then continue; end if; -- remove as políticas antigas ("acesso livre") for r in select policyname from pg_policies where schemaname = 'public' and tablename = t loop execute format('drop policy %I on public.%I', r.policyname, t); end loop; execute format('alter table public.%I enable row level security', t); execute format('create policy "leitura publica" on public.%I for select to anon, authenticated using (true)', t); execute format('create policy "editores inserem" on public.%I for insert to authenticated with check ((select public.pode_editar()))', t); execute format('create policy "editores alteram" on public.%I for update to authenticated using ((select public.pode_editar())) with check ((select public.pode_editar()))', t); execute format('create policy "editores apagam" on public.%I for delete to authenticated using ((select public.pode_editar()))', t); end loop; end $$; -- 7. Fotos dos integrantes (Storage, bucket "fotos") ------------------- do $$ declare r record; begin for r in select policyname from pg_policies where schemaname = 'storage' and tablename = 'objects' and (coalesce(qual, '') || coalesce(with_check, '')) ilike '%fotos%' loop execute format('drop policy %I on storage.objects', r.policyname); end loop; end $$; create policy "fotos leitura" on storage.objects for select to anon, authenticated using (bucket_id = 'fotos'); create policy "fotos editores inserem" on storage.objects for insert to authenticated with check (bucket_id = 'fotos' and (select public.pode_editar())); create policy "fotos editores alteram" on storage.objects for update to authenticated using (bucket_id = 'fotos' and (select public.pode_editar())); create policy "fotos editores apagam" on storage.objects for delete to authenticated using (bucket_id = 'fotos' and (select public.pode_editar()));
